   ConnectServer  {{ .Values.ports.connect }} (original client) / {{ .Values.ports.connectOpenSource }} (open-source client)

   ChatServer     {{ .Values.ports.chat }}
   {{- if .Values.service.hostPort }}

4. Advertised IP (server selection redirect):
   {{- if .Values.resolveIP }}
   RESOLVE_IP={{ .Values.resolveIP }}

   {{- end }}
